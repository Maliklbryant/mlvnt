// Supabase Edge Function: send-consultation-email // Triggered after a successful consultation RPC insert. // Sends two emails via Resend: // 1. Client confirmation // 2. Coach/admin notification to mlvnt2026@gmail.com // // Environment variables required (set in Supabase Dashboard → Edge Functions → Secrets): // RESEND_API_KEY — from resend.com (free tier: 3,000 emails/month) // COACH_EMAIL — defaults to mlvnt2026@gmail.com if not set // FROM_EMAIL — defaults to noreply@mlvnt.com (must be verified domain in Resend) import { serve } from "https://deno.land/std@0.177.0/http/server.ts"; const RESEND_API_URL = "https://api.resend.com/emails"; const COACH_EMAIL = Deno.env.get("COACH_EMAIL") || "mlvnt2026@gmail.com"; const FROM_EMAIL = Deno.env.get("FROM_EMAIL") || "MLVNT "; const RESEND_KEY = Deno.env.get("RESEND_API_KEY"); interface BookingPayload { intake_id: string; first_name: string; last_name: string; email: string; phone?: string; goals?: string[]; level?: string; train_freq?: string; injuries?: string; date_display: string; // "Wednesday, May 28, 2026" time_display: string; // "4:00 PM" } async function sendEmail(to: string, subject: string, html: string): Promise<{ ok: boolean; error?: string }> { if (!RESEND_KEY) { console.error("RESEND_API_KEY not set"); return { ok: false, error: "RESEND_API_KEY not configured" }; } const res = await fetch(RESEND_API_URL, { method: "POST", headers: { "Authorization": `Bearer ${RESEND_KEY}`, "Content-Type": "application/json", }, body: JSON.stringify({ from: FROM_EMAIL, to, subject, html }), }); if (!res.ok) { const body = await res.text(); return { ok: false, error: `Resend ${res.status}: ${body}` }; } return { ok: true }; } function clientEmailHtml(b: BookingPayload): string { const name = [b.first_name, b.last_name].filter(Boolean).join(" ") || "there"; return ` Consultation Confirmed — MLVNT
MLVNT

Consultation Confirmed

You're booked, ${name}.

Your free 30-minute consultation with Malik Bryant has been confirmed.

Date ${b.date_display}
Time ${b.time_display}
Format Virtual / In-person TBC

What happens next

— Malik will review your goals and background before your session.
— You'll receive a calendar invite with connection details.
— Come ready to discuss what you want to achieve.

Questions? Reply to this email or message Malik directly through the MLVNT platform.

Time Moves. So Should You.

MLVNT · New York · mlvnt.com

`; } function coachEmailHtml(b: BookingPayload): string { const name = [b.first_name, b.last_name].filter(Boolean).join(" ") || "Unknown"; const goalsTxt = Array.isArray(b.goals) && b.goals.length ? b.goals.join(", ") : "—"; return ` New Consultation — MLVNT
MLVNT Coach Dashboard

New Consultation Booked

${name} · ${b.date_display}

${[ ["Name", name], ["Email", b.email || "—"], ["Phone", b.phone || "—"], ["Date", b.date_display], ["Time", b.time_display], ["Goals", goalsTxt], ["Level", b.level || "—"], ["Frequency", b.train_freq || "—"], ["Injuries", b.injuries || "None reported"], ].map(([k, v], i, arr) => ` `).join("")}
${k} ${v}

Review this lead in the MLVNT Admin → Consultations section.

MLVNT Coach Portal · Automated notification

`; } serve(async (req: Request) => { // CORS for browser calls if (req.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization", }, }); } if (req.method !== "POST") { return new Response(JSON.stringify({ error: "Method not allowed" }), { status: 405 }); } let body: BookingPayload; try { body = await req.json(); } catch { return new Response(JSON.stringify({ error: "Invalid JSON body" }), { status: 400 }); } console.log("send-consultation-email: received payload for", body.email); const results: Record = {}; // 1. Client confirmation email if (body.email) { const clientName = [body.first_name, body.last_name].filter(Boolean).join(" ") || "there"; const cr = await sendEmail( body.email, `Consultation confirmed — ${body.date_display} at ${body.time_display}`, clientEmailHtml(body) ); results.client_email_sent = cr.ok; if (!cr.ok) { console.error("Client email failed:", cr.error); results.client_email_error = cr.error ?? "unknown"; } else { console.log("Client confirmation sent to", body.email); } } else { results.client_email_sent = false; results.client_email_error = "no client email address"; } // 2. Coach notification email const coachR = await sendEmail( COACH_EMAIL, `New consultation: ${[body.first_name, body.last_name].filter(Boolean).join(" ")} — ${body.date_display}`, coachEmailHtml(body) ); results.coach_email_sent = coachR.ok; if (!coachR.ok) { console.error("Coach email failed:", coachR.error); results.coach_email_error = coachR.error ?? "unknown"; } else { console.log("Coach notification sent to", COACH_EMAIL); } return new Response(JSON.stringify(results), { status: 200, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", }, }); });
